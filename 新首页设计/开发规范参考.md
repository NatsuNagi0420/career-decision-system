## 代码规范

### TypeScript 规范

#### 1. 类型定义

**✅ 推荐**：
```typescript
// 使用 interface 定义对象类型
interface UserInfo {
  id: string
  name: string
  email: string
  avatar?: string
}

// 使用 type 定义联合类型、交叉类型
type JobType = 'intern' | 'fulltime' | 'flexible'
type UserWithRole = UserInfo & { role: string }

// 导出类型
export type { UserInfo, JobType }
```

**❌ 避免**：
```typescript
// 不要使用 any
const data: any = fetchData() // ❌

// 不要省略类型注解（除非能明确推断）
function calculate(a, b) { // ❌
  return a + b
}
```

#### 2. 函数定义

**✅ 推荐**：
```typescript
// 明确参数和返回值类型
function formatDate(date: Date, format: string): string {
  // ...
}

// 使用箭头函数（简短逻辑）
const add = (a: number, b: number): number => a + b

// 使用 async/await
async function fetchUser(id: string): Promise<UserInfo> {
  const response = await api.get(`/users/${id}`)
  return response.data
}
```

**❌ 避免**：
```typescript
// 不要省略返回值类型（复杂函数）
function complexCalculation(data) { // ❌
  // 100 lines of code
  return result
}

// 不要使用 Promise.then（优先 async/await）
function fetchUser(id) { // ❌
  return api.get(`/users/${id}`).then(res => res.data)
}
```

#### 3. 可选链和空值合并

**✅ 推荐**：
```typescript
// 使用可选链
const userName = user?.profile?.name

// 使用空值合并
const displayName = userName ?? 'Anonymous'

// 组合使用
const email = user?.contact?.email ?? 'no-email@example.com'
```

### Vue 3 规范

#### 1. 组合式 API（Composition API）

**✅ 推荐**：
```vue
<script setup lang="ts">
import { ref, computed, onMounted } from 'vue'
import type { UserInfo } from '@/types/models'

// Props 定义
interface Props {
  userId: string
  showAvatar?: boolean
}

const props = withDefaults(defineProps<Props>(), {
  showAvatar: true
})

// Emits 定义
interface Emits {
  (e: 'update', value: UserInfo): void
  (e: 'delete', id: string): void
}

const emit = defineEmits<Emits>()

// 响应式数据
const user = ref<UserInfo | null>(null)
const loading = ref(false)

// 计算属性
const displayName = computed(() => {
  return user.value?.name ?? 'Unknown'
})

// 方法
const fetchUser = async () => {
  loading.value = true
  try {
    user.value = await api.getUser(props.userId)
  } finally {
    loading.value = false
  }
}

// 生命周期
onMounted(() => {
  fetchUser()
})
</script>
```

**❌ 避免**：
```vue
<script>
// 不要使用 Options API（新代码）
export default {
  data() {
    return {
      user: null
    }
  },
  methods: {
    fetchUser() {
      // ...
    }
  }
}
</script>
```

#### 2. 模板规范

**✅ 推荐**：
```vue
<template>
  <div class="user-card">
    <!-- 使用 v-if/v-else-if/v-else -->
    <div v-if="loading" class="loading">加载中...</div>
    <div v-else-if="error" class="error">{{ error }}</div>
    <div v-else-if="user" class="content">
      <!-- 使用简洁的属性绑定 -->
      <img :src="user.avatar" :alt="user.name" />
      <h3>{{ user.name }}</h3>
      
      <!-- 事件处理 -->
      <button @click="handleEdit">编辑</button>
    </div>
    <div v-else class="empty">暂无数据</div>
  </div>
</template>
```

**❌ 避免**：
```vue
<template>
  <!-- 不要在模板中写复杂逻辑 -->
  <div>
    {{ user ? user.name ? user.name.toUpperCase() : 'N/A' : 'Loading' }} <!-- ❌ -->
  </div>
  
  <!-- 不要使用 v-html（XSS 风险） -->
  <div v-html="userInput"></div> <!-- ❌ -->
</template>
```

#### 3. 组件通信

**✅ 推荐**：
```typescript
// 父传子：Props
<ChildComponent :user="user" :show-avatar="true" />

// 子传父：Emits
emit('update', newValue)

// 跨组件：Provide/Inject
// 父组件
provide('theme', theme)

// 子组件
const theme = inject<Ref<string>>('theme')

// 全局状态：Pinia Store
const userStore = useUserStore()
```

## 命名规范

### 文件命名

```
✅ 推荐：
- PascalCase：组件文件
  - UserCard.vue
  - JobList.vue
  - MbtiAssessment.vue

- camelCase：工具函数、Composables
  - useFormValidation.ts
  - formatDate.ts
  - apiClient.ts

- kebab-case：样式文件、配置文件
  - global.css
  - vite.config.ts
  - tsconfig.json

❌ 避免：
- user-card.vue（组件应该用 PascalCase）
- UseFormValidation.ts（Composables 应该用 camelCase）
- GlobalCSS.css（样式应该用 kebab-case）
```

### 变量命名

```typescript
// ✅ 推荐：camelCase
const userName = 'John'
const isLoading = false
const hasPermission = true

// 常量：UPPER_SNAKE_CASE
const MAX_RETRY_COUNT = 3
const API_BASE_URL = 'https://api.example.com'

// 类型/接口：PascalCase
interface UserInfo { }
type JobType = 'intern' | 'fulltime'

// 枚举：PascalCase
enum UserRole {
  Admin = 'admin',
  User = 'user'
}

// ❌ 避免
const UserName = 'John' // 变量不要用 PascalCase
const is_loading = false // 不要用 snake_case
```

### 函数命名

```typescript
// ✅ 推荐：动词开头，camelCase
function fetchUser() { }
function calculateTotal() { }
function validateEmail() { }

// Boolean 返回值：is/has/can 开头
function isValid() { }
function hasPermission() { }
function canEdit() { }

// 事件处理：handle 开头
function handleClick() { }
function handleSubmit() { }

// ❌ 避免
function user() { } // 不清晰
function click() { } // 应该是 handleClick
function check() { } // 应该是 isValid 或 validateXxx
```

### CSS 类名

```css
/* ✅ 推荐：kebab-case，BEM 命名 */
.user-card { }
.user-card__header { }
.user-card__title { }
.user-card--active { }

/* ❌ 避免 */
.UserCard { } /* 不要用 PascalCase */
.user_card { } /* 不要用 snake_case */
.card { } /* 太通用，容易冲突 */
```

## 状态管理规范

### Pinia Store 结构

```typescript
// stores/user.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import type { UserInfo } from '@/types/models'
import { userApi } from '@/api/user'

export const useUserStore = defineStore('user', () => {
  // 1. State
  const userInfo = ref<UserInfo | null>(null)
  const loading = ref(false)
  const error = ref<string | null>(null)

  // 2. Getters
  const isLoggedIn = computed(() => !!userInfo.value)
  const userName = computed(() => userInfo.value?.name ?? 'Guest')

  // 3. Actions
  const fetchUser = async (id: string) => {
    loading.value = true
    error.value = null
    
    try {
      userInfo.value = await userApi.getUser(id)
    } catch (e) {
      error.value = (e as Error).message
      throw e
    } finally {
      loading.value = false
    }
  }

  const updateUser = async (data: Partial<UserInfo>) => {
    if (!userInfo.value) return
    
    const updated = await userApi.updateUser(userInfo.value.id, data)
    userInfo.value = updated
  }

  const logout = () => {
    userInfo.value = null
    // 清理其他状态
  }

  // 4. Return
  return {
    // State
    userInfo,
    loading,
    error,
    
    // Getters
    isLoggedIn,
    userName,
    
    // Actions
    fetchUser,
    updateUser,
    logout
  }
})
```

### Store 使用规范

```typescript
// ✅ 推荐：在组件中使用
<script setup lang="ts">
import { useUserStore } from '@/stores/user'

const userStore = useUserStore()

// 直接访问
console.log(userStore.userName)

// 调用 action
await userStore.fetchUser('123')

// 响应式解构（使用 storeToRefs）
import { storeToRefs } from 'pinia'
const { userName, isLoggedIn } = storeToRefs(userStore)
</script>

// ❌ 避免：直接修改 state
userStore.userInfo = newUser // ❌ 应该通过 action
```

## API 开发规范

### 前端 API 封装

```typescript
// api/user.ts
import { apiClient } from './client'
import type { UserInfo, ApiResponse } from '@/types/api'

export const userApi = {
  // 获取用户信息
  getUser: (id: string) => 
    apiClient.get<UserInfo>(`/users/${id}`),
  
  // 更新用户信息
  updateUser: (id: string, data: Partial<UserInfo>) => 
    apiClient.put<UserInfo>(`/users/${id}`, data),
  
  // 删除用户
  deleteUser: (id: string) => 
    apiClient.delete(`/users/${id}`),
  
  // 用户列表
  listUsers: (params?: { page?: number; pageSize?: number }) => 
    apiClient.get<{ list: UserInfo[]; total: number }>('/users', { params })
}
```

### 后端 API 规范

#### 1. RESTful 设计

```
✅ 推荐：
GET    /api/v1/users           # 获取用户列表
GET    /api/v1/users/:id       # 获取单个用户
POST   /api/v1/users           # 创建用户
PUT    /api/v1/users/:id       # 更新用户（全量）
PATCH  /api/v1/users/:id       # 更新用户（部分）
DELETE /api/v1/users/:id       # 删除用户

❌ 避免：
GET    /api/v1/getUserList     # 不要在 URL 中使用动词
POST   /api/v1/user/delete     # DELETE 操作应该用 DELETE 方法
```

#### 2. 统一响应格式

```typescript
// 成功响应
{
  "code": 0,
  "message": "success",
  "data": {
    "id": "123",
    "name": "John"
  },
  "timestamp": 1234567890
}

// 错误响应
{
  "code": 1001,
  "message": "用户不存在",
  "data": null,
  "timestamp": 1234567890,
  "traceId": "abc-123-def"
}

// 列表响应
{
  "code": 0,
  "message": "success",
  "data": {
    "list": [...],
    "total": 100,
    "page": 1,
    "pageSize": 20
  },
  "timestamp": 1234567890
}
```

#### 3. 错误码规范

```typescript
enum ErrorCode {

  MBTI_NOT_COMPLETED = 1004,

```

## 组件设计原则

### 1. 单一职责原则（SRP）

每个组件只负责一个功能或一个业务场景。

**✅ 好的设计**：
```
UserCard.vue          # 只负责展示用户卡片
UserAvatar.vue        # 只负责展示用户头像
UserActions.vue       # 只负责用户操作按钮
```

**❌ 不好的设计**：
```
UserComponent.vue     # 包含卡片、头像、操作、编辑表单等所有功能
```

### 2. 开闭原则（OCP）

组件应该对扩展开放，对修改关闭。

**✅ 好的设计**：
```vue
<!-- 通过 props 和 slots 扩展功能 -->
<UserCard :user="user">
  <template #actions>
    <CustomActions />
  </template>
</UserCard>
```

**❌ 不好的设计**：
```vue
<!-- 每次新增功能都要修改组件内部代码 -->
<UserCard :user="user" :show-edit="true" :show-delete="true" :show-share="true" />
```

### 3. 组合优于继承

使用组合式 API 和 Composables 而不是组件继承。

**✅ 好的设计**：
```typescript
// 使用 Composables
const { loading, error, data, execute } = useApiRequest(api.getUser)
```

**❌ 不好的设计**：
```typescript
// 使用 mixins（Vue 2 风格）
export default {
  mixins: [apiMixin, loadingMixin]
}
```

### 4. 最小知识原则

组件只需要知道必要的信息，不应该依赖过多的外部状态。

**✅ 好的设计**：
```vue
<!-- 只接收必要的 props -->
<UserCard :user="user" @update="handleUpdate" />
```

**❌ 不好的设计**：
```vue
<!-- 依赖全局状态和多个 store -->
<UserCard />  <!-- 内部直接访问 userStore, authStore, configStore -->
```

## 组件开发流程

### Step 1：需求分析

**问题清单**：
- [ ] 这个组件的核心功能是什么？
- [ ] 需要哪些 props？

```typescript
// ResumeUploader/useResumeParser.ts
import { ref } from 'vue'
import { parseResumeApi } from '@/api/resume'
import type { ResumeParseResult } from './types'

export function useResumeParser() {
  const parsing = ref(false)
  const progress = ref(0)
  const result = ref<ResumeParseResult | null>(null)
  const error = ref<Error | null>(null)

  const parseResume = async (file: File) => {
    parsing.value = true
    progress.value = 0
    error.value = null

    try {
      // 模拟进度
      progress.value = 10

      // 调用 API
      const data = await parseResumeApi(file)
      progress.value = 60

      // 处理结果
      result.value = data
      progress.value = 100
    } catch (e) {
      error.value = e as Error
      throw e
    } finally {
      parsing.value = false
    }
  }

  return {
    parsing,
    progress,
    result,
    error,
    parseResume
  }
}
```

```vue
<!-- DynamicList.vue -->
<template>
  <div class="dynamic-list">
    <div
      v-for="(item, index) in modelValue"
      :key="index"
      class="list-item"
    >
      <slot name="item" :item="item" :index="index" :remove="() => removeItem(index)">
        <!-- 默认渲染 -->
        <el-input v-model="modelValue[index]" />
        <el-button @click="removeItem(index)">
          <el-icon><Delete /></el-icon>
        </el-button>
      </slot>
    </div>

    <el-button type="primary" link @click="addItem">
      <el-icon><Plus /></el-icon>
      {{ addButtonText }}
    </el-button>
  </div>
</template>

<script setup lang="ts" generic="T">
import { Delete, Plus } from '@element-plus/icons-vue'

interface Props {
  modelValue: T[]
  addButtonText?: string
  defaultItem?: T | (() => T)
  maxItems?: number
}

const props = withDefaults(defineProps<Props>(), {
  addButtonText: '添加',
  maxItems: 10
})

interface Emits {
  (e: 'update:modelValue', value: T[]): void
}

const emit = defineEmits<Emits>()

const addItem = () => {
  if (props.modelValue.length >= props.maxItems) {
    return
  }

  const newItem = typeof props.defaultItem === 'function'
    ? props.defaultItem()
    : props.defaultItem ?? ({} as T)

  emit('update:modelValue', [...props.modelValue, newItem])
}

const removeItem = (index: number) => {
  const newValue = [...props.modelValue]
  newValue.splice(index, 1)
  emit('update:modelValue', newValue)
}
</script>

<style scoped>

  display: flex;
  flex-direction: column;
  gap: 12px;
}

  gap: 12px;

}
</style>
```
